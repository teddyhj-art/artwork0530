    function loadStorageData() {
        try {
            const saved = localStorage.getItem('surrealism_workspace_v5_final_leakproof');
            if (saved) {
                const parsed = JSON.parse(saved);
                if (parsed && typeof parsed === 'object') {
                    if (parsed.studentId) state.studentId = parsed.studentId;
                    if (parsed.studentName) state.studentName = parsed.studentName;
                    if (parsed.group && [1, 2, 3, 4].includes(Number(parsed.group))) state.group = Number(parsed.group);
                    if (parsed.subIdx !== undefined) state.subIdx = Number(parsed.subIdx);
                    if (parsed.lang) state.lang = parsed.lang;
                    
                    if (parsed.vault && typeof parsed.vault === 'object') {
                        for (let i = 1; i <= 4; i++) {
                            if (parsed.vault[i] && typeof parsed.vault[i] === 'object') {
                                state.vault[i].common = parsed.vault[i].common || {};
                                state.vault[i].selectQ = parsed.vault[i].selectQ || '';
                                state.vault[i].selectA = parsed.vault[i].selectA || '';
                            }
                        }
                    }
                }
            }
        } catch(e) { console.log("복구 가동"); }
        
        if(document.getElementById('global-student-id')) document.getElementById('global-student-id').value = state.studentId;
        if(document.getElementById('global-student-name')) document.getElementById('global-student-name').value = state.studentName;
        
        selectGroup(state.group);
    }

    function renderPage() {
        const lang = langData[state.lang] || langData['ko']; 
        
        if(document.getElementById('txt-title')) document.getElementById('txt-title').innerText = lang.title;
        if(document.getElementById('txt-subtitle')) document.getElementById('txt-subtitle').innerText = lang.subtitle;
        if(document.getElementById('txt-step1')) document.getElementById('txt-step1').innerText = lang.step1;
        if(document.getElementById('txt-step2')) document.getElementById('txt-step2').innerText = lang.step2;
        if(document.getElementById('txt-step3')) document.getElementById('txt-step3').innerText = lang.step3;
        if(document.getElementById('txt-step4')) document.getElementById('txt-step4').innerText = lang.step4;
        if(document.getElementById('txt-step5')) document.getElementById('txt-step5').innerText = lang.step5;
        if(document.getElementById('txt-btn-hint-common')) document.getElementById('txt-btn-hint-common').innerText = lang.hintBtn;
        if(document.getElementById('txt-btn-hint-selective')) document.getElementById('txt-btn-hint-selective').innerText = lang.hintBtn;
        if(document.getElementById('txt-master-submit-btn')) document.getElementById('txt-master-submit-btn').innerText = lang.masterSubmit;
        if(document.getElementById('txt-btn-copy-all')) document.getElementById('txt-btn-copy-all').innerText = lang.btnCopyAll;
        if(document.getElementById('txt-btn-gform')) document.getElementById('txt-btn-gform').innerText = lang.btnGform;
        
        if(document.getElementById('global-student-id')) document.getElementById('global-student-id').placeholder = lang.placeholderId;
        if(document.getElementById('global-student-name')) document.getElementById('global-student-name').placeholder = lang.placeholderName;
        if(document.getElementById('student-ans-common')) document.getElementById('student-ans-common').placeholder = lang.placeholderAnsCommon;
        if(document.getElementById('student-ans-selective')) document.getElementById('student-ans-selective').placeholder = lang.placeholderAnsSelect;
        if(document.getElementById('custom-q-input')) document.getElementById('custom-q-input').placeholder = lang.customQPlace;

        if(document.getElementById('display-role-title')) {
            document.getElementById('display-role-title').innerText = lang.roles[state.group - 1] || langData['ko'].roles[state.group - 1];
        }

        const qContainer = document.getElementById('sub-q-list');
        if (qContainer) {
            qContainer.innerHTML = '';
            const qList = questionData[state.group] || [];
            qList.forEach((item, index) => {
                const btn = document.createElement('button');
                btn.className = `sub-q-btn ${index === state.subIdx ? 'active-common' : ''}`;
                btn.innerText = item[state.lang] || item['ko'] || ''; 
                btn.onclick = () => selectSubQuestion(index);
                qContainer.appendChild(btn);
            });
        }
    }

    function changeLang(l) {
        state.lang = l;
        document.getElementById('btn-ko').classList.remove('active');
        document.getElementById('btn-en').classList.remove('active');
        document.getElementById('btn-th').classList.remove('active');
        document.getElementById(`btn-${l}`).classList.add('active');
        selectGroup(state.group);
    }

    function selectGroup(groupNum) {
        state.group = groupNum;
        state.subIdx = 0; 
        
        for(let i=1; i<=4; i++) {
            const btn = document.getElementById(`gbtn-${i}`);
            if(btn) {
                if(i === groupNum) btn.classList.add('active');
                else btn.classList.remove('active');
            }
        }
        
        renderPage();
        
        if(document.getElementById('custom-q-input')) document.getElementById('custom-q-input').value = state.vault[groupNum].selectQ || '';
        if(document.getElementById('student-ans-selective')) document.getElementById('student-ans-selective').value = state.vault[groupNum].selectA || '';
        
        selectSubQuestion(0);
    }

    function selectSubQuestion(index) {
        state.subIdx = index;
        const qList = questionData[state.group] || [];
        if(qList[index]) {
            document.getElementById('display-q').innerText = qList[index][state.lang] || qList[index]['ko'];
            document.getElementById('student-ans-common').value = state.vault[state.group].common[index] || '';
        }
        
        const buttons = document.querySelectorAll('.sub-q-btn');
        buttons.forEach((btn, idx) => {
            if(idx === index) btn.classList.add('active-common');
            else btn.classList.remove('active-common');
        });
        
        document.getElementById('hint-display-common').style.display = 'none';
        updateDashboard();
    }

    function updateInputData() {
        state.studentId = document.getElementById('global-student-id').value;
        state.studentName = document.getElementById('global-student-name').value;
        
        state.vault[state.group].common[state.subIdx] = document.getElementById('student-ans-common').value;
        state.vault[state.group].selectQ = document.getElementById('custom-q-input').value;
        state.vault[state.group].selectA = document.getElementById('student-ans-selective').value;
        
        try {
            localStorage.setItem('surrealism_workspace_v5_final_leakproof', JSON.stringify(state));
            const now = new Date();
            document.getElementById('txt-save-time').innerText = `마지막 저장: ${now.toLocaleTimeString()}`;
        } catch(e) {}
        
        updateDashboard();
    }

    function getDashboardText() {
        const lang = langData[state.lang] || langData['ko'];
        let displayStr = `[인적사항]\n학번: ${state.studentId || '(미입력)'} / 이름: ${state.studentName || '(미입력)'}\n\n`;
        displayStr += `[선택 역할]\n#${state.group} - ${lang.roles[state.group - 1]}\n\n`;
        
        displayStr += `[4단계 공통 질문 작성내역]\n`;
        const qList = questionData[state.group] || [];
        qList.forEach((item, index) => {
            const qText = item[state.lang] || item['ko'];
            const ans = state.vault[state.group].common[index] || '';
            displayStr += `Q${index+1}. ${qText}\n👉 답: ${ans || '(미작성)'}\n`;
        });
        
        displayStr += `\n[5단계 선택 질문 작성내역]\n`;
        displayStr += `Q. ${state.vault[state.group].selectQ || '(질문 미입력)'}\n👉 답: ${state.vault[state.group].selectA || '(답변 미작성)'}`;
        return displayStr;
    }

    function updateDashboard() {
        if(document.getElementById('dashboard-view')) {
            document.getElementById('dashboard-view').innerText = getDashboardText();
        }
    }

    function toggleHint(type) {
        const targetId = type === 'common' ? 'hint-display-common' : 'hint-display-selective';
        const box = document.getElementById(targetId);
        if(!box) return;
        
        if(box.style.display === 'block') {
            box.style.display = 'none';
        } else {
            box.style.display = 'block';
            if(type === 'common') {
                const qList = questionData[state.group] || [];
                box.innerText = qList[state.subIdx] ? `💡 AI 예시 방향: ${qList[state.subIdx].a}` : '질문을 먼저 고르세요.';
            } else {
                box.innerText = langData[state.lang].selectiveHint;
            }
        }
    }

    function copyToClipboardFallback(text) {
        const textArea = document.createElement("textarea");
        textArea.value = text;
        textArea.style.top = "0"; textArea.style.left = "0"; textArea.style.position = "fixed"; textArea.style.opacity = "0";
        document.body.appendChild(textArea);
        textArea.focus(); textArea.select();
        try { document.execCommand('copy'); } catch (err) {}
        document.body.removeChild(textArea);
    }

    /* 🚀 핵심 업그레이드: 백그라운드로 안전하게 구글시트 API 전송 */
    function submitFullProject() {
        if(!state.studentId || !state.studentName) {
            alert(state.lang === 'ko' ? '학번과 이름을 먼저 입력해주세요!' : 'Please enter your Student ID and Name!');
            return;
        }

        if(GAS_URL === '여기에_복사한_웹_앱_URL_주소를_넣으세요') {
            alert('설정 안내: 소스코드 최상단에 선생님 구글 웹 앱 URL 주소를 입력해야 전송 기능이 동작합니다.');
            return;
        }

        const organizedText = getDashboardText();

        // fetch 옵션 조율을 통해 로컬 파일 환경 보안 우회 및 제출 신호 송신
        fetch(GAS_URL, {
            method: 'POST',
            mode: 'no-cors', // CORS 제한 무조건 우회 통과 패치
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                studentId: state.studentId,
                studentName: state.studentName,
                group: state.group,
                fullContent: organizedText
            })
        }).then(() => {
            const successMsg = state.lang === 'ko'
                ? '🎉 과제가 교사 전용 스프레드시트로 안전하게 비공개 전송되었습니다!\n다른 친구들은 내가 작성한 내용을 볼 수 없으니 안심하세요.'
                : 'Assignment submitted safely to the teacher sheet!';
            alert(successMsg);
        }).catch((err) => {
            alert('전송 실패: 시스템 환경 오류가 발생했습니다. 수동 복사 기능을 사용해 주세요.');
        });
    }

    function copyDashboardText() {
        const organizedText = getDashboardText();
        if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(organizedText).then(() => {
                alert('📋 전체 내용이 클립보드에 복사되었습니다.');
            }).catch(() => {
                copyToClipboardFallback(organizedText);
                alert('📋 전체 내용이 클립보드에 복사되었습니다.');
            });
        } else {
            copyToClipboardFallback(organizedText);
            alert('📋 전체 내용이 클립보드에 복사되었습니다.');
        }
    }
